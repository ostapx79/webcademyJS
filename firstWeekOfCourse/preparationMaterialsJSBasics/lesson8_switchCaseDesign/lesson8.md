### 08 Конструкций Switch Case
```javascript
var mark = 4;

switch (mark) {
    case 5:
        console.log("Отлично! Хорошая работа!");
        break;
    case 4:
        console.log("Хороший результат!");
        break;
    case 3:
        console.log("Можно было и лучше!");
        break;
    default:
        console.log("Такой оценки не существует!");
}
```
